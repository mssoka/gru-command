#!/usr/bin/env node
/**
 * Dispatch client for this lane's tracked reviewer jobs (read the briefing
 * files from the review-packet/briefings directory). Credentials stay in
 * this process; only the summarized dispatch response is printed.
 *
 * Usage: node dispatch-reviewers.mjs <briefing-file> <job_id> <title>
 */
import { readFileSync } from 'node:fs';
import { request as httpRequest } from 'node:http';
import { join } from 'node:path';

function readConfigField(section, field) {
  const text = readFileSync(join(process.env.HOME ?? '', '.gru-command', 'config.toml'), 'utf-8');
  const sectionMatch = new RegExp(`^\\[${section}\\]\\s*$`, 'm').exec(text);
  if (sectionMatch === null) throw new Error(`config section [${section}] not found`);
  const rest = text.slice(sectionMatch.index + sectionMatch[0].length);
  const nextSection = /^\[/m.exec(rest);
  const body = nextSection === null ? rest : rest.slice(0, nextSection.index);
  const fieldMatch = new RegExp(`^\\s*${field}\\s*=\\s*(?:"([^"]*)"|'([^']*)'|([^#\\n]+))`, 'm').exec(body);
  if (fieldMatch === null) throw new Error(`config field ${field} not found in [${section}]`);
  return (fieldMatch[1] ?? fieldMatch[2] ?? fieldMatch[3] ?? '').trim();
}

const [briefingPath, jobId, title] = process.argv.slice(2);
if (briefingPath === undefined || jobId === undefined || title === undefined) {
  process.stderr.write('usage: node dispatch-reviewers.mjs <briefing-file> <job_id> <title>\n');
  process.exit(2);
}

const port = Number(readConfigField('server', 'port'));
const token = readConfigField('auth', 'token');
const briefing = readFileSync(briefingPath, 'utf-8');
const body = JSON.stringify({
  job_id: jobId,
  repo_path: '/Users/moses/code/gru-command',
  title,
  briefing,
});

const response = await new Promise((resolve, reject) => {
  const req = httpRequest(
    {
      host: '127.0.0.1', port, path: '/api/dispatch', method: 'POST',
      headers: { 'content-type': 'application/json', 'content-length': Buffer.byteLength(body), authorization: `Bearer ${token}` },
      timeout: 300_000,
    },
    (res) => {
      let raw = '';
      res.setEncoding('utf-8');
      res.on('data', (c) => { raw += c; });
      res.on('end', () => resolve({ status: res.statusCode, body: raw }));
    },
  );
  req.on('error', reject);
  req.on('timeout', () => req.destroy(new Error('dispatch transport timeout')));
  req.write(body);
  req.end();
});

process.stdout.write(`POST /api/dispatch (${jobId}) -> HTTP ${String(response.status)}\n`);
process.stdout.write(`${response.body.slice(0, 1200)}\n`);
process.exit(response.status !== undefined && response.status < 300 ? 0 : 1);
